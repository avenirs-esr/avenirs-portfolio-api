package fr.avenirsesr.portfolio.association.domain.service;

import fr.avenirsesr.portfolio.association.domain.exception.AssociationLockedException;
import fr.avenirsesr.portfolio.association.domain.model.Association;
import fr.avenirsesr.portfolio.association.domain.model.EAssociationType;
import fr.avenirsesr.portfolio.association.domain.port.input.AssociationCheckService;
import fr.avenirsesr.portfolio.association.domain.port.input.AssociationService;
import fr.avenirsesr.portfolio.common.error.domain.model.enums.EErrorCode;
import fr.avenirsesr.portfolio.student.progress.declared.activity.domain.model.DeclaredActivity;
import fr.avenirsesr.portfolio.student.progress.declared.activity.domain.port.input.DeclaredActivityService;
import java.util.List;
import java.util.UUID;
import lombok.RequiredArgsConstructor;

@RequiredArgsConstructor
public class AssociationCheckServiceImpl implements AssociationCheckService {
  private final AssociationService associationService;
  private final DeclaredActivityService declaredActivityService;

  @Override
  public boolean areAssociationsUnlockedOf(UUID sourceId, Class<?> sourceClass) {
    var associations =
        associationService.getAllOf(sourceId, sourceClass, EAssociationType.getAllBy(sourceClass));

    return areAssociationsUnlocked(associations);
  }

  @Override
  public boolean areAssociationsUnlocked(List<Association> associations) {
    return areDeclaredActivityAssociationsUnlocked(associations);
  }

  @Override
  public void checkAssociationsLockedOf(UUID sourceId, Class<?> sourceClass) {
    var associations =
        associationService.getAllOf(sourceId, sourceClass, EAssociationType.getAllBy(sourceClass));

    checkAssociationsLocked(associations);
  }

  @Override
  public void checkAssociationsLocked(List<Association> associations) {
    if (!areDeclaredActivityAssociationsUnlocked(associations)) {
      throw new AssociationLockedException(EErrorCode.DECLARED_ACTIVITY_LOCKED);
    }
  }

  private boolean areDeclaredActivityAssociationsUnlocked(List<Association> associations) {
    var declaredActivityIds = getDeclaredActivityIds(associations);

    if (declaredActivityIds.isEmpty()) {
      return true;
    }

    var declaredActivities =
        declaredActivityService.findAllDeclaredActivitiesByIds(declaredActivityIds);

    return declaredActivities.stream().noneMatch(declaredActivityService::isSubmittedOrFinished);
  }

  private List<UUID> getDeclaredActivityIds(List<Association> associations) {
    return associations.stream()
        .filter(association -> association.getAssociationType().contains(DeclaredActivity.class))
        .map(
            association ->
                association
                    .getAssociationType()
                    .idExtractorFor(DeclaredActivity.class)
                    .apply(association))
        .distinct()
        .toList();
  }
}
