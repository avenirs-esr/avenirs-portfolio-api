package fr.avenirsesr.portfolio.cgu.infrastructure.adapter.repository;

import fr.avenirsesr.portfolio.cgu.domain.model.CguAcceptance;
import fr.avenirsesr.portfolio.cgu.domain.port.output.repository.CguAcceptanceRepository;
import fr.avenirsesr.portfolio.cgu.infrastructure.adapter.mapper.CguAcceptanceMapper;
import fr.avenirsesr.portfolio.cgu.infrastructure.adapter.model.CguAcceptanceEntity;
import fr.avenirsesr.portfolio.common.data.infrastructure.adapter.repository.GenericJpaRepositoryAdapter;
import java.util.Optional;
import java.util.UUID;
import org.springframework.stereotype.Repository;

@Repository
public class CguAcceptanceDatabaseRepository
    extends GenericJpaRepositoryAdapter<CguAcceptance, CguAcceptanceEntity>
    implements CguAcceptanceRepository {

  private final CguAcceptanceJpaRepository cguAcceptanceJpaRepository;

  public CguAcceptanceDatabaseRepository(CguAcceptanceJpaRepository cguAcceptanceJpaRepository) {
    super(
        cguAcceptanceJpaRepository,
        cguAcceptanceJpaRepository,
        CguAcceptanceEntity.class,
        CguAcceptanceMapper.INSTANCE);
    this.cguAcceptanceJpaRepository = cguAcceptanceJpaRepository;
  }

  @Override
  public Optional<CguAcceptance> findLastByUser(UUID userId) {
    return cguAcceptanceJpaRepository
        .findFirstByUserIdOrderByAcceptedAtDesc(userId)
        .map(CguAcceptanceMapper.INSTANCE::toDomain);
  }

  @Override
  public Optional<CguAcceptance> findByUserAndCgu(UUID userId, UUID cguId) {
    return cguAcceptanceJpaRepository
        .findByUserIdAndCguId(userId, cguId)
        .map(CguAcceptanceMapper.INSTANCE::toDomain);
  }
}
