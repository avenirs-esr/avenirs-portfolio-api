package fr.avenirsesr.portfolio.cgu.domain.service;

import fr.avenirsesr.portfolio.cgu.domain.exception.CguNotFoundException;
import fr.avenirsesr.portfolio.cgu.domain.model.CguAcceptance;
import fr.avenirsesr.portfolio.cgu.domain.port.input.CguAcceptanceService;
import fr.avenirsesr.portfolio.cgu.domain.port.output.client.CguClient;
import fr.avenirsesr.portfolio.cgu.domain.port.output.repository.CguAcceptanceRepository;
import fr.avenirsesr.portfolio.common.cgu.application.adapter.dto.CguDTO;
import fr.avenirsesr.portfolio.common.data.domain.model.User;
import fr.avenirsesr.portfolio.shared.domain.port.input.LoggedInUserService;
import java.util.Optional;
import lombok.AllArgsConstructor;
import lombok.extern.slf4j.Slf4j;

@Slf4j
@AllArgsConstructor
public class CguAcceptanceServiceImpl implements CguAcceptanceService {
  private final CguClient cguClient;
  private final CguAcceptanceRepository cguAcceptanceRepository;
  private final LoggedInUserService loggedInUserService;

  @Override
  public CguAcceptance acceptCurrent() {
    User user = loggedInUserService.getLoggedInUser();
    CguDTO current = cguClient.getLatest().orElseThrow(CguNotFoundException::new);

    return cguAcceptanceRepository
        .findByUserAndCgu(user.getId(), current.id())
        .orElseGet(() -> saveAcceptance(user, current));
  }

  @Override
  public Optional<CguAcceptance> getLastAcceptance() {
    return cguAcceptanceRepository.findLastByUser(loggedInUserService.getLoggedInUser().getId());
  }

  private CguAcceptance saveAcceptance(User user, CguDTO current) {
    CguAcceptance acceptance =
        cguAcceptanceRepository.save(CguAcceptance.create(user, current.id()));
    log.info("Terms of use version {} accepted by user [{}]", current.version(), user.getId());

    return acceptance;
  }
}
