package fr.avenirsesr.portfolio.cgu.domain.service;

import fr.avenirsesr.portfolio.cgu.domain.data.CguAcceptanceData;
import fr.avenirsesr.portfolio.cgu.domain.exception.CguNotFoundException;
import fr.avenirsesr.portfolio.cgu.domain.port.input.CguAcceptanceService;
import fr.avenirsesr.portfolio.cgu.domain.port.output.client.CguClient;
import fr.avenirsesr.portfolio.common.cgu.application.adapter.dto.CguDTO;
import fr.avenirsesr.portfolio.common.data.domain.model.User;
import fr.avenirsesr.portfolio.shared.domain.port.input.LoggedInUserService;
import fr.avenirsesr.portfolio.user.domain.port.output.repository.UserRepository;
import java.time.Instant;
import java.time.temporal.ChronoUnit;
import java.util.Optional;
import lombok.AllArgsConstructor;
import lombok.extern.slf4j.Slf4j;

@Slf4j
@AllArgsConstructor
public class CguAcceptanceServiceImpl implements CguAcceptanceService {
  private final CguClient cguClient;
  private final UserRepository userRepository;
  private final LoggedInUserService loggedInUserService;

  @Override
  public CguAcceptanceData acceptCurrent() {
    User user = loggedInUserService.getLoggedInUser();
    CguDTO current = cguClient.getLatest().orElseThrow(CguNotFoundException::new);

    if (current.id().equals(user.getAcceptedCguId())) {
      return acceptanceOf(user);
    }

    user.setAcceptedCguId(current.id());
    user.setAcceptedCguAt(Instant.now().truncatedTo(ChronoUnit.MICROS));
    userRepository.save(user);
    log.info("Terms of use version {} accepted by user [{}]", current.version(), user.getId());

    return acceptanceOf(user);
  }

  @Override
  public Optional<CguAcceptanceData> getLastAcceptance() {
    User user = loggedInUserService.getLoggedInUser();

    return Optional.ofNullable(user.getAcceptedCguId())
        .map(cguId -> new CguAcceptanceData(cguId, user.getAcceptedCguAt()));
  }

  private static CguAcceptanceData acceptanceOf(User user) {
    return new CguAcceptanceData(user.getAcceptedCguId(), user.getAcceptedCguAt());
  }
}
