  @Override
  public Activity create(
      UUID id,
      Staff author,
      String title,
      EActivityThematic thematic,
      String summary,
      String description,
      String recommendedCompletionContexts,
      LocalDate startDate,
      LocalDate endDate,
      boolean enableReflection,
      int traceAllowedAssociations,
      int feedbackAllowedIterations,
      List<String> links) {
    requireNotBlankAndMaxLength("title", title, TITLE_LENGTH);
    requireNotNull("thematic", thematic);
    requireNotBlankAndMaxLength("summary", summary, SUMMARY_LENGTH);
    requireNotBlankAndEnrichedMaxLength("description", description, RICH_DESCRIPTION_LENGTH);
    requireNotBlankAndMaxLength(
        "recommendedCompletionContexts",
        recommendedCompletionContexts,
        ACTIVITY_RECOMMENDED_COMPLETION_CONTEXTS);
    if ((startDate == null) != (endDate == null)) {
      throw new ActivityDatesException();
    }

    if (startDate != null) {
      validateDateOrder(startDate, endDate);
    }

    var activity =
        Activity.create(
            id,
            author,
            title,
            thematic,
            summary,
            description,
            recommendedCompletionContexts,
            startDate,
            endDate,
            enableReflection,
            traceAllowedAssociations,
            feedbackAllowedIterations,
            null,
            links,
            List.of(),
            List.of(),
            List.of());
    activityRepository.save(activity);
    activityAuthorizationService.grantAllRights(activity.getId(), author);
    return activity;
  }

  @Override
  public Activity createSubActivity(
      UUID id,
      UUID previousActivityId,
      String title,
      String description,
      String recommendedCompletionContexts,
      LocalDate startDate,
      LocalDate endDate,
      List<String> links) {
    requireNotBlankAndMaxLength("title", title, TITLE_LENGTH);
    requireNotBlankAndEnrichedMaxLength("description", description, RICH_DESCRIPTION_LENGTH);
    validateOptionalTextMaxLength(
        "recommendedCompletionContexts",
        recommendedCompletionContexts,
        ACTIVITY_RECOMMENDED_COMPLETION_CONTEXTS);
    if ((startDate == null) != (endDate == null)) {
      throw new ActivityDatesException();
    }
    if (startDate != null) {
      validateDateOrder(startDate, endDate);
    }

    var previousActivity =
        activityRepository.findById(previousActivityId).orElseThrow(ActivityNotFoundException::new);
    var mainActivity = getMainActivityOf(previousActivity);

    return activityRepository.save(
        Activity.createSubActivity(
            id,
            mainActivity.getAuthor(),
            mainActivity.getId(),
            previousActivity.getId(),
            title,
            description,
            recommendedCompletionContexts,
            startDate,
            endDate,
            links,
            List.of()));
  }

  @Override
  public Activity publish(UUID activityDraftId) {
    var staff = loggedInUserService.getLoggedInStaff();
    var draft =
        activityDraftRepository
            .findById(activityDraftId)
            .orElseThrow(ActivityDraftNotFoundException::new);
    requireMainActivity(draft);
    activityAuthorizationService.checkDraftRight(staff, activityDraftId, EActivityRight.PUBLISH);

    if (draft.getThematic().isEmpty()) {
      throw new FieldValidationException(
          EErrorCode.NOT_BLANK, "thematic must be defined to publish an activity");
    }

    if (draft.getSummary().isEmpty()) {
      throw new FieldValidationException(
          EErrorCode.NOT_BLANK, "summary must be defined to publish an activity");
    }

    if (draft.getDescription().isEmpty()) {
      throw new FieldValidationException(
          EErrorCode.NOT_BLANK, "description must be defined to publish an activity");
    }

    draft.getAssessment().ifPresent(ActivityAssessment::validate);

    var subActivityDrafts = activityDraftRepository.findAllSubActivitiesOf(activityDraftId);
    if (subActivityDrafts.stream()
        .anyMatch(subActivityDraft -> subActivityDraft.getDescription().isEmpty())) {
      throw new FieldValidationException(
          EErrorCode.NOT_BLANK, "description must be defined to publish a sub-activity");
    }

    var publishedActivity = activityRepository.findById(activityDraftId);

    Activity activity =
        publishedActivity.orElse(
            Activity.create(
                draft.getId(),
                draft.getAuthor(),
                draft.getTitle(),
                draft.getThematic().orElseThrow(),
                draft.getSummary().orElseThrow(),
                draft.getDescription().orElseThrow(),
                draft.getRecommendedCompletionContexts().orElse(null),
                draft.getStartDate().orElse(null),
                draft.getEndDate().orElse(null),
                draft.isEnableReflection(),
                draft.getTraceAllowedAssociations(),
                draft.getFeedbackAllowedIterations(),
                draft.getBanner().orElse(null),
                draft.getLinks(),
                draft.getFiles(),
                draft.getTargetInstitutionIds(),
                draft.getTargetGroupIds()));
    activity.setNational(draft.isNational());
    activity.setAssessment(draft.getAssessment().orElse(null));

    var removedFileIds = new ArrayList<UUID>();
    if (publishedActivity.isPresent()) {
      removedFileIds.addAll(removedFileIdsOf(activity, draft));
      var enrolledDeclaredActivities = declaredActivityService.getEnrolledStudents(activity);
      var updatedFields = updateActivity(activity, draft, !enrolledDeclaredActivities.isEmpty());
      activity.setStatus(EActivityStatus.PUBLISHED);
      notifyActivityUpdated(updatedFields, enrolledDeclaredActivities);
    }

    var savedActivity = activityRepository.save(activity);
    activityAuthorizationService.applyDraftAuthorizationsToActivity(savedActivity.getId());
    removedFileIds.addAll(publishSubActivities(activityDraftId, subActivityDrafts));

    removeSubActivityDrafts(subActivityDrafts);
    activityDraftRepository.removeFromDatabase(draft);
    removedFileIds.forEach(fileResourceService::delete);
    return savedActivity;
  }

  private List<UUID> publishSubActivities(UUID activityId, List<ActivityDraft> subActivityDrafts) {
    if (subActivityDrafts.isEmpty()) {
      return List.of();
    }
    var publishedById =
        activityRepository.findAllSubActivitiesOf(activityId).stream()
            .collect(Collectors.toMap(Activity::getId, Function.identity()));

    var removedFileIds = new ArrayList<UUID>();
    var subActivities = new ArrayList<Activity>();
    for (var subActivityDraft : subActivityDrafts) {
      var published = publishedById.get(subActivityDraft.getId());
      if (published == null) {
        subActivities.add(subActivityOf(subActivityDraft));
        continue;
      }
      removedFileIds.addAll(removedFileIdsOf(published, subActivityDraft));
      updateActivity(published, subActivityDraft, true);
      published.attachAsSubActivity(
          subActivityDraft.getMainActivityId().orElseThrow(),
          subActivityDraft.getPreviousActivityId().orElseThrow());
      subActivities.add(published);
    }
    activityRepository.saveAll(subActivities);
    return removedFileIds;
  }

  private Activity subActivityOf(ActivityDraft subActivityDraft) {
    return Activity.createSubActivity(
        subActivityDraft.getId(),
        subActivityDraft.getAuthor(),
        subActivityDraft.getMainActivityId().orElseThrow(),
        subActivityDraft.getPreviousActivityId().orElseThrow(),
        subActivityDraft.getTitle(),
        subActivityDraft.getDescription().orElseThrow(),
        subActivityDraft.getRecommendedCompletionContexts().orElse(null),
        subActivityDraft.getStartDate().orElse(null),
        subActivityDraft.getEndDate().orElse(null),
        subActivityDraft.getLinks(),
        subActivityDraft.getFiles());
  }

  private List<UUID> removedFileIdsOf(Activity activity, ActivityDraft draft) {
    var draftFileIds = draft.getFiles().stream().map(File::getId).collect(Collectors.toSet());
    return activity.getFiles().stream()
        .map(File::getId)
        .filter(id -> !draftFileIds.contains(id))
        .toList();
  }

  private record FieldSync<T>(
      EActivityUpdatableField field, T currentValue, T newValue, Consumer<T> setter) {
    boolean applyIfChanged() {
      if (Objects.equals(currentValue, newValue)) {
        return false;
      }
      setter.accept(newValue);
      return true;
    }
  }

  private List<EActivityUpdatableField> updateActivity(
      Activity activity, ActivityDraft draft, boolean hasEnrolledStudents) {
    var syncs =
        List.of(
            new FieldSync<>(
                ACTIVITY_TITLE, activity.getTitle(), draft.getTitle(), activity::setTitle),
            new FieldSync<>(
                SUMMARY,
                activity.getSummary(),
                draft.getSummary().orElse(null),
                activity::setSummary),
            new FieldSync<>(
                DESCRIPTION,
                activity.getDescription(),
                draft.getDescription().orElse(null),
                activity::setDescription),
            new FieldSync<>(
                RECOMMENDED_COMPLETION_CONTEXTS,
                activity.getRecommendedCompletionContexts().orElse(null),
                draft.getRecommendedCompletionContexts().orElse(null),
                activity::setRecommendedCompletionContexts),
            new FieldSync<>(
                THEMATIC,
                activity.getThematic(),
                draft.getThematic().orElse(null),
                activity::setThematic),
            new FieldSync<>(
                BANNER,
                activity.getBanner().orElse(null),
                draft.getBanner().orElse(null),
                activity::setBanner),
            new FieldSync<>(
                FILES_AND_LINKS,
                new HashSet<>(activity.getLinks()),
                new HashSet<>(draft.getLinks()),
                links -> activity.setLinks(links.stream().toList())),
            new FieldSync<>(
                FILES_AND_LINKS, activity.getFiles(), draft.getFiles(), activity::setFiles));

    var updatedFields =
        syncs.stream().filter(FieldSync::applyIfChanged).map(FieldSync::field).distinct().toList();

    activity.setStartDate(draft.getStartDate().orElse(null));
    activity.setEndDate(draft.getEndDate().orElse(null));
    activity.setTargetInstitutionIds(draft.getTargetInstitutionIds());
    activity.setTargetGroupIds(draft.getTargetGroupIds());
    activity.setNational(draft.isNational());

    if (!hasEnrolledStudents) {
      activity.setTraceAllowedAssociations(draft.getTraceAllowedAssociations());
      activity.setFeedbackAllowedIterations(draft.getFeedbackAllowedIterations());
      activity.setEnableReflection(draft.isEnableReflection());
    }

    return updatedFields;
  }

  private void notifyActivityUpdated(
      List<EActivityUpdatableField> updatedFields, List<DeclaredActivity> declaredActivities) {
    if (declaredActivities.isEmpty() || updatedFields.isEmpty()) {
      return;
    }
    notificationService.notifyAll(
        declaredActivities.stream()
            .map(
                declaredActivity ->
                    new ActivityUpdatedNotification(
                        declaredActivity.getStudent().getUser(), declaredActivity, updatedFields))
            .toList());
  }

  @Override
  public Activity unpublish(UUID activityId) {
    var staff = loggedInUserService.getLoggedInStaff();
    var activity =
        activityRepository.findById(activityId).orElseThrow(ActivityNotFoundException::new);
    requireMainActivity(activity);
    activityAuthorizationService.checkRight(staff, activityId, EActivityRight.PUBLISH);
    if (activity.getStatus() == EActivityStatus.UNPUBLISHED) {
      throw new ActivityUnpublishedException();
    }
    activity.setStatus(EActivityStatus.UNPUBLISHED);
    return activityRepository.save(activity);
  }

  @Override
  public void deleteDraft(UUID activityDraftId) {
    var staff = loggedInUserService.getLoggedInStaff();
    var draft =
        activityDraftRepository
            .findById(activityDraftId)
            .orElseThrow(ActivityDraftNotFoundException::new);
    requireMainActivity(draft);
    activityAuthorizationService.checkDraftRight(staff, activityDraftId, EActivityRight.EDIT);

    removeSubActivityDrafts(activityDraftRepository.findAllSubActivitiesOf(activityDraftId));
    activityDraftRepository.removeFromDatabase(draft);
    activityAuthorizationService.deleteAllOfDraft(activityDraftId);
    log.info("Deleted activity draft with id: {}", activityDraftId);
  }

  @Override
  public void deleteActivityDefinitively(UUID activityId) {
    var activity =
        activityRepository.findById(activityId).orElseThrow(ActivityNotFoundException::new);
    requireMainActivity(activity);
    var draft = activityDraftRepository.findById(activityId);
    var subActivities = activityRepository.findAllSubActivitiesOf(activityId);
    var subActivityDrafts = activityDraftRepository.findAllSubActivitiesOf(activityId);
    var fileIds =
        Stream.of(
                fileIdsOf(activity.getBanner().orElse(null), activity.getFiles()),
                draft
                    .map(it -> fileIdsOf(it.getBanner().orElse(null), it.getFiles()))
                    .orElseGet(Stream::empty),
                subActivities.stream()
                    .flatMap(it -> fileIdsOf(it.getBanner().orElse(null), it.getFiles())),
                subActivityDrafts.stream()
                    .flatMap(it -> fileIdsOf(it.getBanner().orElse(null), it.getFiles())))
            .flatMap(ids -> ids)
            .distinct()
            .toList();
    var subActivityIds =
        Stream.concat(
                subActivities.stream().map(Activity::getId),
                subActivityDrafts.stream().map(ActivityDraft::getId))
            .distinct()
            .toList();

    declaredActivityService.deleteAllOfActivity(activity);
    activityViewRepository.deleteAllByActivityId(activityId);
    subActivityIds.forEach(activityViewRepository::deleteAllByActivityId);
    removeSubActivityDrafts(subActivityDrafts);
    draft.ifPresent(activityDraftRepository::removeFromDatabase);
    removeSubActivities(subActivities);
    activityRepository.removeFromDatabase(activity);
    activityAuthorizationService.deleteAllOfActivity(activityId);
    activityAuthorizationService.deleteAllOfDraft(activityId);
    fileIds.forEach(fileResourceService::delete);

    log.info("Definitively deleted activity {} and every element linked to it", activityId);
  }

  private Stream<UUID> fileIdsOf(File banner, List<File> files) {
    return Stream.concat(Stream.ofNullable(banner), files.stream()).map(File::getId);
  }

  private static <T extends ActivityNode> List<T> tailFirst(List<T> orderedSubActivities) {
    return orderedSubActivities.reversed();
  }

  @Override
  public ActivityPresentationData getActivityPresentation(EActivityStatus activityStatus, UUID id) {
    return switch (activityStatus) {
      case PUBLISHED, UNPUBLISHED -> {
        Activity activity =
            activityRepository.findById(id).orElseThrow(ActivityNotFoundException::new);
        var mainActivity = getMainActivityOf(activity);
        var declaredActivity = declaredActivityService.getByActivity(mainActivity);
        var declaredActivityId = declaredActivity.map(DeclaredActivity::getId);
        var declaredActivityStatus =
            declaredActivity.map(declaredActivityService::getDeclaredActivityStatus);
        activityViewRepository.recordView(
            activity.getId(), loggedInUserService.getLoggedInStudent().getId());
        yield ActivityPresentationDataMapper.toData(
            activity,
            mainActivity.getStatus(),
            declaredActivityId.orElse(null),
            declaredActivityStatus.orElse(null),
            FileDataMapper.mapFileData(
                activity.getBanner(), FileStorageConstants.DEFAULT_COVER_FILE_URL),
            getSubActivityOverviews(activity.getId()));
      }
      case DRAFT -> {
        var draft =
            activityDraftRepository.findById(id).orElseThrow(ActivityDraftNotFoundException::new);
        yield ActivityPresentationDataMapper.toData(
            draft,
            FileDataMapper.mapFileData(
                draft.getBanner(), FileStorageConstants.DEFAULT_COVER_FILE_URL),
            getDraftSubActivityOverviews(draft.getId()));
      }
    };
  }

  @Override
  public ActivityDashboardData getActivityDashboard(UUID activityId) {
    loggedInUserService.getLoggedInStaff();
    var activity =
        activityRepository.findById(activityId).orElseThrow(ActivityNotFoundException::new);
    requireMainActivity(activity);

    return new ActivityDashboardData(
        activityViewRepository.countUniqueViews(activityId),
        declaredActivityService.countEnrolledStudents(activity),
        declaredActivityService.countUnsubscriptionsSince(
            activity, Instant.now().minus(DURATION_FOR_RECENT_UNSUBSCRIPTIONS)),
        countInactiveStudents(activityId, activity));
  }

  private int countInactiveStudents(UUID activityId, Activity activity) {
    var enrolledStudentIds = declaredActivityService.getEnrolledStudentIds(activity);
    if (enrolledStudentIds.isEmpty()) {
      return 0;
    }
    var since = Instant.now().minus(DURATION_FOR_STUDENT_INACTIVITY);
    return enrolledStudentIds.size()
        - activityViewRepository.countViewersSince(activityId, enrolledStudentIds, since);
  }

  @Override
  public List<InactiveStudentData> getInactiveStudents(UUID activityId) {
    loggedInUserService.getLoggedInStaff();
    var activity =
        activityRepository.findById(activityId).orElseThrow(ActivityNotFoundException::new);
    requireMainActivity(activity);

    var since = Instant.now().minus(DURATION_FOR_STUDENT_INACTIVITY);
    var inactiveDeclaredActivities =
        declaredActivityService.getEnrolledStudentsNotViewedSince(activity, since);
    if (inactiveDeclaredActivities.isEmpty()) {
      return List.of();
    }

    var lastViewedAtByStudent =
        activityViewRepository.findLastViewedAtByStudents(
            activityId,
            inactiveDeclaredActivities.stream()
                .map(declaredActivity -> declaredActivity.getStudent().getId())
                .toList());

    return inactiveDeclaredActivities.stream()
        .map(
            declaredActivity ->
                new InactiveStudentData(
                    declaredActivity.getStudent(),
                    declaredActivity.getCreatedAt(),
                    lastViewedAtByStudent.get(declaredActivity.getStudent().getId())))
        .toList();
  }

  @Override
  public Map<EActivityThematic, List<Activity>> getActivityNavigation() {
    return activityRepository.findAll().stream()
        .filter(a -> !a.isSubActivity())
        .filter(a -> a.getStatus() != EActivityStatus.UNPUBLISHED)
        .collect(
            Collectors.groupingBy(
                Activity::getThematic,
                () -> new EnumMap<>(EActivityThematic.class),
                Collectors.toList()));
  }

  private PagedResult<ActivityStaffOverviewData> withLeaders(
      PagedResult<ActivityStaffOverviewData> pagedActivities) {
    var activityLeaders =
        activityAuthorizationService.getLeadersByActivityIds(
            idsOf(pagedActivities.content(), false));
    var draftLeaders =
        activityAuthorizationService.getLeadersByDraftIds(idsOf(pagedActivities.content(), true));
    return new PagedResult<>(
        pagedActivities.content().stream()
            .map(
                a -> {
                  var leaders = isDraft(a) ? draftLeaders : activityLeaders;
                  return a.withLeaders(leaders.getOrDefault(a.activityId(), List.of()));
                })
            .toList(),
        pagedActivities.pageInfo());

  @Override

    var mainDraft = getDraftWithRight(activityDraftId, EActivityRight.EDIT);
    requireMainActivity(mainDraft);

    }
    requireNotBlankAndMaxLength("title", title, TITLE_LENGTH);

    var subActivities = activityDraftRepository.findAllSubActivitiesOf(activityDraftId);
    var previousActivityId =
        subActivities.isEmpty() ? activityDraftId : subActivities.getLast().getId();

  @Override
  public void updateSubActivityDraftsOrder(UUID activityDraftId, List<UUID> orderedSubActivityIds) {
    var mainDraft = getDraftWithRight(activityDraftId, EActivityRight.EDIT);
    requireMainActivity(mainDraft);

    var subActivitiesById =
        activityDraftRepository.findAllSubActivitiesOf(activityDraftId).stream()
            .collect(Collectors.toMap(ActivityDraft::getId, Function.identity()));
    if (orderedSubActivityIds == null
        || orderedSubActivityIds.size() != subActivitiesById.size()
        || !subActivitiesById.keySet().equals(new HashSet<>(orderedSubActivityIds))) {
      throw new SubActivityOrderMismatchException();
    }

    var reordered = new ArrayList<ActivityDraft>();
    var previousActivityId = activityDraftId;
    for (var subActivityId : orderedSubActivityIds) {
      var subActivity = subActivitiesById.get(subActivityId);
      subActivity.attachAsSubActivity(activityDraftId, previousActivityId);
      reordered.add(subActivity);
      previousActivityId = subActivityId;
    }
    if (!reordered.isEmpty()) {
      activityDraftRepository.saveAll(reordered);
    }
    log.info("Reordered the sub-activities of activity draft with id: {}", activityDraftId);
  }

  @Override
  public ActivityDraft updateActivityDraft(
      UUID id,
      String title,
      EActivityThematic thematic,
      String summary,
      String description,
      String recommendedCompletionContexts,
      LocalDate startDate,
      LocalDate endDate,
      Integer traceAllowedAssociations,
      Integer feedbackAllowedIterations,
      Boolean enableReflection,
      List<String> links,
      Boolean enableCompletionPeriod,
      List<UUID> targetInstitutionIds,
      List<UUID> targetGroupIds,
      Boolean national,
      ActivityAssessment assessment) {
    var loggedInStaff = loggedInUserService.getLoggedInStaff();
    var draft =
        activityDraftRepository.findById(id).orElseThrow(ActivityDraftNotFoundException::new);
    activityAuthorizationService.checkDraftRight(
        loggedInStaff, draft.getMainActivityId().orElse(id), EActivityRight.EDIT);

    if (draft.isSubActivity()) {
      rejectIfProvided("thematic", thematic);
      rejectIfProvided("summary", summary);
      rejectIfProvided("traceAllowedAssociations", traceAllowedAssociations);
      rejectIfProvided("feedbackAllowedIterations", feedbackAllowedIterations);
      rejectIfProvided("enableReflection", enableReflection);
      rejectIfProvided("targetInstitutionIds", targetInstitutionIds);
      rejectIfProvided("targetGroupIds", targetGroupIds);
      rejectIfProvided("national", national);
      rejectIfProvided("assessment", assessment);
    }

    validateTargetPerimeter(loggedInStaff, targetInstitutionIds, targetGroupIds);
    validateTargetingUpdate(id, targetInstitutionIds, targetGroupIds);

    validateOptionalTextMaxLength("summary", summary, SUMMARY_LENGTH);
    validateOptionalTextMaxLength("description", description, RICH_DESCRIPTION_LENGTH);
    validateOptionalTextMaxLength(
        "recommendedCompletionContexts",
        recommendedCompletionContexts,
        ACTIVITY_RECOMMENDED_COMPLETION_CONTEXTS);
    if (startDate != null) {
      validateDateOrder(startDate, endDate);
    }
    if ((startDate == null) != (endDate == null)
        || (Boolean.TRUE.equals(enableCompletionPeriod) && startDate == null)) {
      throw new ActivityDatesException();
    }
    if (Boolean.TRUE.equals(enableCompletionPeriod)) {
      validatePeriodAgainstRelatedActivities(draft, startDate, endDate);
    }
    if (assessment != null) {
      assessment.validate();
    }

    if (title != null) {
      requireNotBlankAndMaxLength("title", title, TITLE_LENGTH);
      draft.setTitle(title);
    }
    if (thematic != null) draft.setThematic(thematic);
    if (summary != null) draft.setSummary(summary);
    if (description != null) draft.setDescription(description);
    if (recommendedCompletionContexts != null)
      draft.setRecommendedCompletionContexts(recommendedCompletionContexts);
    if (Boolean.TRUE.equals(enableCompletionPeriod)) {
      draft.setStartDate(startDate);
      draft.setEndDate(endDate);
    } else if (Boolean.FALSE.equals(enableCompletionPeriod)) {
      draft.setStartDate(null);
      draft.setEndDate(null);
    }

    if (!hasEnrolledStudents(draft)) {
      if (traceAllowedAssociations != null)
        draft.setTraceAllowedAssociations(traceAllowedAssociations);
      if (feedbackAllowedIterations != null)
        draft.setFeedbackAllowedIterations(feedbackAllowedIterations);
      if (enableReflection != null) draft.setEnableReflection(enableReflection);
    }

    if (links != null && !links.isEmpty()) {
      links.forEach(
          link -> {
            requireNotBlankAndMaxLength("link", link, LINK_LENGTH);
            validateUrl(link);
          });
      draft.setLinks(links);
    }

    if (targetInstitutionIds != null) draft.setTargetInstitutionIds(targetInstitutionIds);
    if (targetGroupIds != null) draft.setTargetGroupIds(targetGroupIds);
    if (national != null) draft.setNational(national);

    if (assessment != null) {
      draft.setAssessment(assessment.isEmpty() ? null : assessment);
    }

    var updatedDraft = activityDraftRepository.save(draft);
    log.info("Updated activity draft with id: {}", id);
    return updatedDraft;
  }

  private static void rejectIfProvided(String fieldName, Object value) {
    if (value != null) {
      throw new SubActivityFieldNotAllowedException(fieldName);
    }
  }

  private void validatePeriodAgainstRelatedActivities(
      ActivityDraft draft, LocalDate startDate, LocalDate endDate) {
    if (draft.isSubActivity()) {
      var mainDraft =
          activityDraftRepository
              .findById(draft.getMainActivityId().orElseThrow())
              .orElseThrow(ActivityDraftNotFoundException::new);
      if (mainDraft.getStartDate().isPresent()
          && !isWithinPeriod(
              startDate,
              endDate,
              mainDraft.getStartDate().orElseThrow(),
              mainDraft.getEndDate().orElseThrow())) {
        throw new SubActivityPeriodOutsideMainActivityException();
      }
      return;
    }

    var coversSubActivities =
        activityDraftRepository.findAllSubActivitiesOf(draft.getId()).stream()
            .filter(subActivityDraft -> subActivityDraft.getStartDate().isPresent())
            .allMatch(
                subActivityDraft ->
                    isWithinPeriod(
                        subActivityDraft.getStartDate().orElseThrow(),
                        subActivityDraft.getEndDate().orElseThrow(),
                        startDate,
                        endDate));
    if (!coversSubActivities) {
      throw new ActivityPeriodExcludesSubActivitiesException();
    }
  }

  private static boolean isWithinPeriod(
      LocalDate startDate, LocalDate endDate, LocalDate periodStart, LocalDate periodEnd) {
    return !startDate.isBefore(periodStart) && !endDate.isAfter(periodEnd);
  }

  private void validateTargetPerimeter(
      Staff staff, List<UUID> targetInstitutionIds, List<UUID> targetGroupIds) {
    boolean hasInstitutionTargets = targetInstitutionIds != null && !targetInstitutionIds.isEmpty();
    boolean hasGroupTargets = targetGroupIds != null && !targetGroupIds.isEmpty();

    if ((hasInstitutionTargets || hasGroupTargets)
        && !accessClient.staffHasAccess(
            eppnOf(staff.getId()), targetInstitutionIds, targetGroupIds)) {
      throw new ActivityTargetNotAccessibleException();
    }
  }

  private StudentAccessScope studentScope(Student student) {
    return accessClient.getStudentScope(eppnOf(student.getId()));
  }

  private String eppnOf(UUID userId) {
    return userPrincipalRepository.findEppnByUserId(userId).orElseThrow(UserNotFoundException::new);
  }

  private void validateTargetingUpdate(
      UUID activityId, List<UUID> targetInstitutionIds, List<UUID> targetGroupIds) {
    var publishedActivity =
        activityRepository
            .findById(activityId)
            .filter(activity -> activity.getStatus() == EActivityStatus.PUBLISHED);
    if (publishedActivity.isEmpty()) {
      return;
    }

    var activity = publishedActivity.get();
    if (targetInstitutionIds != null
            && !targetInstitutionIds.containsAll(activity.getTargetInstitutionIds())
        || targetGroupIds != null && !targetGroupIds.containsAll(activity.getTargetGroupIds())) {
      throw new ActivityTargetRemovalNotAllowedException();
    }
  }

  @Override
  public ActivityDraft createDraftFromActivity(UUID activityId) {
    var staff = loggedInUserService.getLoggedInStaff();
    var activity =
        activityRepository.findById(activityId).orElseThrow(ActivityNotFoundException::new);
    requireMainActivity(activity);
    activityAuthorizationService.checkRight(staff, activityId, EActivityRight.EDIT);

    var activityDraft = activityDraftRepository.findById(activityId);

    if (activityDraft.isPresent()) {
      return activityDraft.get();
    }

    var savedDraft = activityDraftRepository.save(draftOf(activity));
    var subActivities = activityRepository.findAllSubActivitiesOf(activityId);
    if (!subActivities.isEmpty()) {
      activityDraftRepository.saveAll(subActivities.stream().map(this::draftOf).toList());
    }
    activityAuthorizationService.copyActivityAuthorizationsToDraft(activityId);
    if (!hasEnrolledStudents(activity)) {
      removeSubActivities(subActivities);
      activityRepository.removeFromDatabase(activity);
    }
    return savedDraft;
  }

  private ActivityDraft draftOf(Activity activity) {
    var draft =
        ActivityDraft.toDomain(
            activity.getId(),
            activity.getCreatedAt(),
            activity.getUpdatedAt(),
            activity.getTitle(),
            activity.getAuthor(),
            activity.getThematic(),
            activity.getSummary(),
            activity.getDescription(),
            activity.getRecommendedCompletionContexts().orElse(null),
            activity.getStartDate().orElse(null),
            activity.getEndDate().orElse(null),
            activity.getTraceAllowedAssociations(),
            activity.getFeedbackAllowedIterations(),
            activity.isEnableReflection(),
            activity.getBanner().orElse(null),
            activity.getLinks(),
            activity.getFiles(),
            activity.getTargetInstitutionIds(),
            activity.getTargetGroupIds(),
            activity.isNational(),
            activity.getAssessment().orElse(null));
    activity
        .getMainActivityId()
        .ifPresent(
            mainActivityId ->
                draft.attachAsSubActivity(
                    mainActivityId, activity.getPreviousActivityId().orElseThrow()));
    return draft;
  }

  @Override
  public ActivityDraft duplicateActivity(UUID activityId, String title) {
    var staff = loggedInUserService.getLoggedInStaff();
    requireNotBlankAndMaxLength("title", title, TITLE_LENGTH);

    var now = Instant.now();
    var publishedSource = activityRepository.findById(activityId);

    if (publishedSource.isPresent()) {
      var source = publishedSource.get();
      requireMainActivity(source);
      var duplicate =
          saveDuplicate(duplicateOf(source, UUID.randomUUID(), staff, title, now), staff);
      saveDuplicatedSubActivities(
          activityRepository.findAllSubActivitiesOf(activityId),
          duplicate.getId(),
          (subActivity, id) -> duplicateOf(subActivity, id, staff, subActivity.getTitle(), now));
      return duplicate;
    }

    var draftSource =
        activityDraftRepository.findById(activityId).orElseThrow(ActivityNotFoundException::new);
    requireMainActivity(draftSource);
    var duplicate =
        saveDuplicate(duplicateOf(draftSource, UUID.randomUUID(), staff, title, now), staff);
    saveDuplicatedSubActivities(
        activityDraftRepository.findAllSubActivitiesOf(activityId),
        duplicate.getId(),
        (subActivity, id) -> duplicateOf(subActivity, id, staff, subActivity.getTitle(), now));
    return duplicate;
  }

  private ActivityDraft saveDuplicate(ActivityDraft duplicate, Staff staff) {
    var savedDuplicate = activityDraftRepository.save(duplicate);
    activityAuthorizationService.grantAllDraftRights(savedDuplicate.getId(), staff);
    return savedDuplicate;
  }

  private <T extends ActivityNode> void saveDuplicatedSubActivities(
      List<T> orderedSourceSubActivities,
      UUID duplicateMainActivityId,
      BiFunction<T, UUID, ActivityDraft> duplicator) {
    if (orderedSourceSubActivities.isEmpty()) {
      return;
    }
    var duplicates = new ArrayList<ActivityDraft>();
    var previousActivityId = duplicateMainActivityId;
    for (var sourceSubActivity : orderedSourceSubActivities) {
      var duplicate = duplicator.apply(sourceSubActivity, UUID.randomUUID());
      duplicate.attachAsSubActivity(duplicateMainActivityId, previousActivityId);
      previousActivityId = duplicate.getId();
      duplicates.add(duplicate);
    }
    activityDraftRepository.saveAll(duplicates);
  }
